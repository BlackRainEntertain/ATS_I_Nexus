// ==UserScript==
// @name         Meta Firefox experiment
// @version      1.1
// @description  Extrahiert Meta AI Chat-Antworten für die barrierefreie Audio-Resonanz im Nexus-System.
// @author       Architekt & Gee
// @match        https://www.facebook.com/*
// @match        *://www.facebook.com*
// @match        *://www.meta.ai/*
// @match        *://www.messenger.com*
// @grant        GM_xmlhttpRequest
// @grant        unsafeWindow
// @run-at       document-end
// @connect      localhost
// @connect      *
// @connect      127.0.0.1
// @noframes     false
// ==/UserScript==


(function() {
    'use strict';
    const ENDPOINT = "http://127.0.0.1:8002/";
    let lastSent = "";
    let debounceTimer = null;

    const startObserving = () => {
        // Wir suchen den Container, der NICHT dein Eingabefeld ist
        const chatWall = document.querySelector('div[role="main"]') || document.body;

        const observer = new MutationObserver(() => {
            // Wir suchen alle potenziellen Text-Container
            const elements = document.querySelectorAll('p, div[dir="auto"]');
            let storyFragments = [];

            // Wir identifizieren den letzten Block, um den "Ursprung" zu finden
            for (let i = elements.length - 1; i >= 0; i--) {
                const el = elements[i];
                const style = window.getComputedStyle(el);

                // Wir suchen nur nach Meta-Elementen (Links, nicht editierbar)
                const isMeta = style.textAlign !== "right" &&
                               !el.isContentEditable &&
                               !el.closest('[role="textbox"]') &&
                               !el.closest('.x1yc453h');

                if (isMeta && el.innerText.trim().length > 1) {
                    // Wir haben einen Teil der Nachricht gefunden.
                    // Jetzt suchen wir alle Geschwister-Elemente im selben Nachrichten-Block
                    const messageGroup = el.closest('[role="row"], div[data-testid="message_container"], .x1n2onr6');

                    if (messageGroup) {
                        // Sammle allen Text aus dieser EINEN Nachrichtengruppe
                        const allParts = messageGroup.querySelectorAll('p, div[dir="auto"]');
                        allParts.forEach(p => {
                            const partText = p.innerText.trim();
                            if (partText.length > 0 && !storyFragments.includes(partText)) {
                                storyFragments.push(partText);
                            }
                        });
                    } else {
                        // Fallback: Wenn keine Gruppe gefunden, nimm nur diesen einen Absatz
                        storyFragments.push(el.innerText.trim());
                    }
                    break; // Wir haben die aktuellste Nachrichtengruppe gefunden
                }
            }

            const fullText = storyFragments.join("\n\n"); // Absätze mit Leerzeile trennen

            if (fullText && fullText !== lastSent && fullText !== "...") {
                clearTimeout(debounceTimer);
                debounceTimer = setTimeout(() => {
                    if (fullText !== lastSent) {
                        lastSent = fullText;
                        console.log("[NEXUS] Ganze Story erfasst (" + fullText.length + " Zeichen)");
                        send(fullText);
                    }
                }, 4000); // Etwas mehr Zeit geben, falls Meta langsam tippt
            }
        });


        observer.observe(chatWall, { childList: true, subtree: true, characterData: true });
    };

    function send(t) {
        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({ text: t, ts: Date.now() }),
            headers: { "Content-Type": "application/json" },
            onload: () => console.log("[NEXUS OK]")
        });
    }

    setTimeout(startObserving, 5000);
})();
