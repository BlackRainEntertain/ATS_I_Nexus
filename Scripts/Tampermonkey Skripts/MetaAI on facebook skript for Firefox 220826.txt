// ==UserScript==
// @name         Meta.ai (ON FACEBOOK) Firefox Perfekt 260426
// @version      1.0
// @description  Extrahiert Meta AI Chat-Antworten für die barrierefreie Audio-Resonanz im Nexus-System.
// @author       Architekt & Gee
// @match        https://www.facebook.com/*
// @match        *://www.facebook.com*
// @match        *://www.meta.ai/*
// @match        *://www.messenger.com*
// @grant        GM_xmlhttpRequest
// @grant        unsafeWindow
// @run-at       document-end
// @connect      localhost
// @connect      *
// @connect      127.0.0.1
// @noframes     false
// ==/UserScript==


(function() {
    'use strict';
    const ENDPOINT = "http://127.0.0.1:8002/";
    let lastSent = "";
    let debounceTimer = null;
    let isSending = false; // Der neue Türsteher

    // --- NEU: DER META-FOKUS-MAGNET (v2.0) ---
    window.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            const metaInput = document.querySelector('[data-lexical-editor="true"]');
            if (metaInput) {
                metaInput.focus();
                console.log("[NEXUS] Meta-Fokus magnetisch auf Lexical-Editor gezogen!");
                metaInput.style.border = "2px solid #0084ff";
                setTimeout(() => { metaInput.style.border = "none"; }, 500);
            }
        }
    });

    // --- LARYNX-PARAMETER (Cyberpunk-Safe) ---
    const PATIENCE_TIME = 4000;
    const DROSS_LIMIT = 300000; // 5 Min Standby
    const IDLE_LIMIT = 900000; // 15 Min Deep Sleep
    let lastActivity = Date.now();
    let mutationObserver = null;
    let idleCheckerInterval = null;

    const getThrottleMode = () => {
        const idleMs = Date.now() - lastActivity;
        if (idleMs >= IDLE_LIMIT) return 'deep_sleep';
        if (document.hidden || idleMs >= DROSS_LIMIT) return 'standby';
        return 'active';
    };

    const updateActivity = () => {
        lastActivity = Date.now();
        if (getThrottleMode() !== 'deep_sleep') startObserver();
    };

    const harvestMeta = () => {
        // 1. Namen aus dem Header holen
        const nameSpan = document.querySelector('.xxymvpz.x1dyh7pn');
        let senderName = "NEXUS_PARTNER";
        if (nameSpan && nameSpan.innerText.trim()) {
            senderName = nameSpan.innerText.trim();
        }

        // 2. DAS CHIRURGISCHE INTERFACE-ZIELFERNROHR (Deine Idee!):
        // Wir holen uns ausschliesslich die echten Nachrichten-Sprechblasen über role="presentation"
        const messageBubbles = document.querySelectorAll('div[role="presentation"].html-div.xdj266r.x14z9mp.xat24cr.x1lziwak');
        if (messageBubbles.length === 0) return;

        // 3. Wir scannen von der allerletzten Sprechblase rückwärts nach oben
        for (let i = messageBubbles.length - 1; i >= 0; i--) {
            const bubble = messageBubbles[i];

            // SPERRE 1: Liegt die Blase in deiner offenen Tippbox? -> Überspringen!
            if (bubble.closest('[role="textbox"], [contenteditable="true"], [data-lexical-editor="true"]')) {
                continue;
            }

            // SPERRE 2: Ist das deine eigene gesendete Nachricht? (Enthält deine x3jgonx-Klasse) -> Überspringen!
            if (bubble.querySelector('.x3jgonx') || bubble.classList.contains('x3jgonx')) {
                continue;
            }

            // SPERRE 3: Keine versteckten em-Regieanweisungen von Meta miternten
            if (bubble.querySelector('em') && bubble.innerText.trim().startsWith('...')) {
                // Falls die Blase NUR aus einer Regieanweisung besteht, überspringen wir sie
                if (bubble.querySelectorAll('div.xdj266r').length <= 1) continue;
            }

            // --- 4. TEXT EXTRAHIEREN ---
            let storyFragments = [];

            // Wir holen uns die echten Textzeilen innerhalb dieser bereinigten KI-Sprechblase
            bubble.querySelectorAll('div.xdj266r.x14z9mp.xat24cr.x1lziwak').forEach(div => {
                // Versteckte Gedanken-Zeilen im Inneren ausblenden
                if (div.querySelector('em')) return;

                const txt = div.innerText.trim();
                if (txt && !storyFragments.includes(txt)) {
                    storyFragments.push(txt);
                }
            });

            const fullText = storyFragments.join("\n");

            // --- 5. ABSENDEN AN DEN NEXUS ---
            if (fullText && fullText !== lastSent && fullText.length > 2 && !fullText.endsWith("...")) {
                send(fullText, senderName);
                break; // Ernte erfolgreich für diesen Durchgang, Schleife beenden!
            }
        }
    };


    const startObserver = () => {
        if (window.nexusObserverMeta) {
            try { window.nexusObserverMeta.disconnect(); } catch(e) {}
            window.nexusObserverMeta = null;
        }

        if (mutationObserver) return;

        mutationObserver = new MutationObserver(() => {
            clearTimeout(debounceTimer);
            // Meta braucht etwas mehr Geduld (4s)
            debounceTimer = setTimeout(() => {
                console.log("[NEXUS Meta] Text stabil. Ernte...");
                harvestMeta();
            }, 5000);
        });

        mutationObserver.observe(document.body, { childList: true, subtree: true });
        window.nexusObserverMeta = mutationObserver;
        console.log("[NEXUS Meta] Observer aktiv.");
    };


    const stopObserver = () => {
        if (mutationObserver) {
            mutationObserver.disconnect();
            mutationObserver = null;
        }
        console.log("[NEXUS Meta] Observer gestoppt (Deep-Sleep)");
    };

    const startIdleChecker = () => {
        if (idleCheckerInterval) return;
        idleCheckerInterval = setInterval(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') {
                stopObserver();
            } else if (document.hidden) {
                // Nur ernten, wenn wir NICHT im Tab sind
                harvestMeta();
                if (!mutationObserver) startObserver();
            }
        }, 5000); // 5 Sekunden Herzschlag
    };


    const wakeUpCall = () => {
        lastActivity = Date.now();
        // isSending = false; <-- GELÖSCHT: Wir brechen keine laufenden Sendevorgänge ab.

        if (window.nexusObserverMeta) {
            try { window.nexusObserverMeta.disconnect(); } catch(e) {}
            window.nexusObserverMeta = null;
        }
        mutationObserver = null;

        startObserver();
        // harvestMeta() hier entfernt, der Observer übernimmt beim Fokus-Wechsel
        console.log("[NEXUS Meta] System reaktiviert.");
    };


    const fullWake = () => {
        if (getThrottleMode() !== 'active') wakeUpCall();
        else updateActivity();
    };

    function send(t, name) {
        if (!t || isSending) return;

        // Identitäts-Check auf die ersten 100 Zeichen
        // (Ignoriert Meta-UI-Müll am Ende der Nachricht)
        const currentHash = btoa(encodeURIComponent(t.slice(0, 100)));
        if (currentHash === lastSent) return;

        isSending = true;
        lastSent = currentHash; // Wir speichern den Hash als Referenz

        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({ text: t, ts: Date.now(), sender: name, hash: currentHash }),
            headers: { "Content-Type": "application/json" },
            timeout: 7000,
            onload: () => {
                // Kurze Sperre für Snap-Resonanz
                setTimeout(() => { isSending = false; }, 1000);
                console.log("[NEXUS Meta] Übermittelt.");
            },
            onerror: () => { isSending = false; },
            ontimeout: () => { isSending = false; }
        });
    }


    // --- EVENT LISTENER (Das neue Förmchen) ---
    document.addEventListener('visibilitychange', () => { if (!document.hidden) wakeUpCall(); });
    window.addEventListener('focus', wakeUpCall);
    document.addEventListener('mousedown', () => {
        if (getThrottleMode() !== 'active') wakeUpCall();
        else lastActivity = Date.now();
    });
    document.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            if (getThrottleMode() !== 'active') wakeUpCall();
            const metaInput = document.querySelector('[data-lexical-editor="true"]');
            if (metaInput) { metaInput.focus(); }
            return;
        }
        if (getThrottleMode() !== 'active') wakeUpCall();
        else lastActivity = Date.now();
    });

    // --- SYSTEM-START (Die stehen bleiben müssen) ---
    startObserver();
    startIdleChecker();
})();
