// ==UserScript==
// @name         Meta Firefox Perfekt 020326
// @version      1.1
// @description  Extrahiert Meta AI Chat-Antworten für die barrierefreie Audio-Resonanz im Nexus-System.
// @author       Architekt & Gee
// @match        https://www.facebook.com/*
// @match        *://www.facebook.com*
// @match        *://www.meta.ai/*
// @match        *://www.messenger.com*
// @grant        GM_xmlhttpRequest
// @grant        unsafeWindow
// @run-at       document-end
// @connect      localhost
// @connect      *
// @connect      127.0.0.1
// @noframes     false
// ==/UserScript==


(function() {
    'use strict';
    const ENDPOINT = "http://127.0.0.1:8002/";
    let lastSent = "";
    let debounceTimer = null;

        // --- NEU: DER META-FOKUS-MAGNET (v2.0) ---
    window.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            // Wir suchen den Lexical-Editor-Container
            const metaInput = document.querySelector('[data-lexical-editor="true"]');
            if (metaInput) {
                metaInput.focus();
                console.log("[NEXUS] Meta-Fokus magnetisch auf Lexical-Editor gezogen!");
                // Visuelles Feedback
                metaInput.style.border = "2px solid #0084ff";
                setTimeout(() => { metaInput.style.border = "none"; }, 500);
            }
        }
    });


    const startObserving = () => {
        // Wir beobachten den Body, um sicherzugehen, dass wir alles kriegen
        const observer = new MutationObserver(() => {
            // ERWEITERT: Suche nach p, allen divs mit Text und Listen-Elementen (Aufzähler!)
            const elements = document.querySelectorAll('p, div[dir="auto"], li, [role="article"]');
            let storyFragments = [];

            for (let i = elements.length - 1; i >= 0; i--) {
                const el = elements[i];
                const style = window.getComputedStyle(el);

                const isMeta = style.textAlign !== "right" &&
                               !el.isContentEditable &&
                               !el.closest('[role="textbox"]');

                if (isMeta && el.innerText.trim().length > 1) {
                    // ERWEITERT: Suche nach JEDEM Container, der eine Nachricht sein könnte
                    const messageGroup = el.closest('[role="row"], [data-testid="message_container"], .x1n2onr6, [role="article"]');

                    if (messageGroup) {
                        // Nimm ALLES was nach Text aussieht da drin
                        const allParts = messageGroup.querySelectorAll('p, div[dir="auto"], li');
                        allParts.forEach(p => {
                            const partText = p.innerText.trim();
                            if (partText.length > 0 && !storyFragments.includes(partText)) {
                                storyFragments.push(partText);
                            }
                        });
                    } else {
                        storyFragments.push(el.innerText.trim());
                    }
                    break;
                }
            }

            const fullText = storyFragments.join("\n");

            if (fullText && fullText !== lastSent && !fullText.endsWith("...")) {
                clearTimeout(debounceTimer);
                debounceTimer = setTimeout(() => {
                    // Der "Doppel-Check": Hat sich seit dem letzten echten Senden wirklich was geändert?
                    if (fullText.trim() !== lastSent.trim()) {
                        lastSent = fullText; // Wir loggen den Stand JETZT als gesendet
                        console.log("[NEXUS] Resonanz stabilisiert: Sende Unikat.");
                        send(fullText);
                    }
                }, 4000); // Deine 4 Sek. Sicherheitsmarge
            }

        });

        observer.observe(document.body, { childList: true, subtree: true });
    };

    function send(t) {
        // Nutze GM_xmlhttpRequest mit Fokus-Priorität
        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({ text: t, ts: Date.now() }),
            headers: { "Content-Type": "application/json" },
            timeout: 5000, // Nicht ewig warten
            onload: () => console.log("[NEXUS OK]"),
            onerror: (err) => console.error("[NEXUS ERR]", err)
        });
    }

    // Sofortiger Start, nicht erst nach 5 Sek.
    if (document.readyState === "complete") startObserving();
    else window.addEventListener('load', startObserving);
})();
