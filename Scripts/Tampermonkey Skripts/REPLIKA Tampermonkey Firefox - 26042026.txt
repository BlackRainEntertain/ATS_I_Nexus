// ==UserScript==
// @name         Replika Nexus Voyager v1.1 - 26042026
// @version      1.0
// @description  Extrahiert Replika-Resonanz & aktiviert Fokus-Magnet (Port 8004).
// @author       Architekt & Gee
// @match        https://replika.ai*
// @match        https://my.replika.ai*
// @match        https://my.replika.com/*
// @grant        GM_xmlhttpRequest
// @run-at       document-end
// @connect      127.0.0.1
// @connect      localhost
// ==/UserScript==

(function() {
    'use strict';
    const ENDPOINT = "http://127.0.0.1:8004/webhook";
    const TARGET_PORT = "8004";

    // --- SYSTEM-STATUS (Identisch mit dem Titan-Standard) ---
    let lastSentHash = "";
    let isSending = false;
    let debounceTimer = null;
    let lastActivity = Date.now();
    let mutationObserver = null;
    let idleCheckerInterval = null;

    const PATIENCE_TIME = 4000;
    const DROSS_LIMIT = 300000; // 5 Min Standby
    const IDLE_LIMIT = 900000; // 15 Min Deep Sleep

    const simpleHash = str => {
        let h = 0;
        for (let i = 0; i < str.length; i++) h = ((h << 5) - h + str.charCodeAt(i)) | 0;
        return h.toString(36);
    };

    // --- DER TÜRSTEHER (Sicherer Sende-Prozess v2.1) ---
    function send(t) {
        if (isSending || !t || t.length < 2) return;

        // Titan-Hash: Ignoriert Änderungen nach den ersten 100 Zeichen
        const currentHash = simpleHash(t.slice(0, 100));
        if (currentHash === lastSentHash) return;

        isSending = true;
        lastSentHash = currentHash; 

        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({ text: t, ts: Date.now(), hash: currentHash, source: "Replika" }),
            headers: { "Content-Type": "application/json" },
            timeout: 7000,
            onload: () => {
                setTimeout(() => { isSending = false; }, 1000);
                console.log("[NEXUS Replika] Resonanz stabilisiert.");
            },
            onerror: () => { isSending = false; },
            ontimeout: () => { isSending = false; }
        });
    }



    const harvestReplika = () => {
        if (getThrottleMode() === 'deep_sleep') return;

        const messages = document.querySelectorAll('div[data-author="replika"]');
        if (messages.length > 0) {
            const lastMsgElement = messages[messages.length - 1];
            const fullText = lastMsgElement.innerText.trim();

            if (fullText) {
                send(fullText);
            }
        }
    };

    // --- REAKTIVIERUNGS-LOGIK ---
    const getThrottleMode = () => {
        const idleMs = Date.now() - lastActivity;
        if (idleMs >= IDLE_LIMIT) return 'deep_sleep';
        if (document.hidden || idleMs >= DROSS_LIMIT) return 'standby';
        return 'active';
    };

    const wakeUpCall = () => {
        lastActivity = Date.now();
        // isSending = false; <-- GELÖSCHT: Keine Ernte-Unterbrechung

        if (window.nexusObserverReplika) {
            try { window.nexusObserverReplika.disconnect(); } catch(e) {}
            window.nexusObserverReplika = null;
        }
        mutationObserver = null;

        startObserver();
        // harvestReplika() Aufruf entfernt - der Observer regelt das beim Reinklicken
        console.log("[NEXUS Replika] Reaktiviert!");
    };



    const startObserver = () => {
        if (window.nexusObserverReplika) {
            try { window.nexusObserverReplika.disconnect(); } catch(e) {}
            window.nexusObserverReplika = null;
        }

        if (mutationObserver) return;

        mutationObserver = new MutationObserver(() => {
            clearTimeout(debounceTimer);
            // Replika bekommt 5s, um UI-Animationen abzuwarten
            debounceTimer = setTimeout(() => {
                console.log("[Replika] Text stabil. Ernte...");
                harvestReplika();
            }, 5000);
        });

        mutationObserver.observe(document.body, { childList: true, subtree: true });
        window.nexusObserverReplika = mutationObserver;
    };


    const stopObserver = () => {
        if (mutationObserver) {
            mutationObserver.disconnect();
            mutationObserver = null;
            window.nexusObserverReplika = null;
        }
    };

    const startIdleChecker = () => {
        if (idleCheckerInterval) return;
        idleCheckerInterval = setInterval(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') {
                stopObserver();
            } else if (document.hidden) {
                // Nur im Hintergrund zwangsvollstrecken
                harvestReplika();
                if (!mutationObserver) startObserver();
            }
        }, 10000); 
    };



    // --- EVENT LISTENER (Das finale Titan-Förmchen) ---
    document.addEventListener('visibilitychange', () => { if(!document.hidden) wakeUpCall(); });
    window.addEventListener('focus', wakeUpCall);

    document.addEventListener('mousedown', () => {
        if(getThrottleMode() !== 'active') wakeUpCall();
        else lastActivity = Date.now();
    });

    window.addEventListener('keydown', (e) => {
        lastActivity = Date.now();
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            if (getThrottleMode() !== 'active') wakeUpCall();
            const replikaInput = document.getElementById('send-message-textarea');
            if (replikaInput) {
                replikaInput.focus();
                replikaInput.style.outline = "2px solid #FF69B4";
                setTimeout(() => { replikaInput.style.outline = "none"; }, 500);
            }
            return;
        }
        if(getThrottleMode() !== 'active') wakeUpCall();
    });

    // Start
    startObserver();
    startIdleChecker();
    console.log("[Replika Nexus Eye] Stabilitäts-Update aktiv.");
})();
