// ==UserScript==
// @name         Grok Nexus Eye v1 - Port 8006 Vivaldi browser 250426
// @version      1.0
// @description  Grok on X - Nexus Eye v17 experimentell
// @author       Bre + Grok
// @match        https://grok.x.ai/*
// @match        https://grok.com/*
// @match        https://grok.ai/*
// @match        https://x.com/i/grok/*
// @include      https://*.x.ai/*
// @include      https://*.grok.ai/*
// @include      https://*.grok.com/*
// @include      https://x.com/i/grok*
// @grant        GM_xmlhttpRequest
// @connect      127.0.0.1
// @connect      127.0.0.1:8006
// @run-at       document-end
// ==/UserScript==

(function () {
    'use strict';

    // 1. HARDWARE- & NETZWERK-ANKER
    const TARGET_PORT = "8006";
    const ENDPOINT = `http://127.0.0.1:${TARGET_PORT}/webhook`;
    const MIN_LENGTH = 10;

    // 2. DROSSLUNGS-PARAMETER (IDENTISCH MIT GEE)
    const PATIENCE_TIME = 3000;
    const DROSS_LIMIT = 5 * 60 * 1000;
    const IDLE_LIMIT = 15 * 60 * 1000;

    // 3. SYSTEM-STATUS
    let lastActivity = Date.now();
    let lastSentHash = "";
    let debounceTimer = null;
    let mutationObserver = null;
    let idleCheckerInterval = null;
    let isPaused = false;
    let isSending = false; // Der neue Türsteher für Grok


    const simpleHash = str => {
        let h = 0;
        for (let i = 0; i < str.length; i++) h = ((h << 5) - h + str.charCodeAt(i)) | 0;
        return h.toString(36);
    };

    const send = (text) => {
        // TÜRSTEHER-CHECK: Wenn wir senden, pausiert sind oder Text zu kurz ist -> Abbruch
        if (isSending || !text || text.length < MIN_LENGTH || isPaused) return;

        const hash = simpleHash(text);
        if (hash === lastSentHash) return;

        isSending = true; // Riegel vorschieben
        lastSentHash = hash;

        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({
                text: text,
                ts: Date.now(),
                hash: hash,
                source: "grok"
            }),
            headers: { "Content-Type": "application/json; charset=UTF-8" },
            timeout: 7000,
            onload: () => {
                // Nach 1 Sekunde wieder freigeben
                setTimeout(() => { isSending = false; }, 1000);
            },
            onerror: () => {
                isSending = false;
                lastSentHash = "";
                console.warn("[NEXUS Grok] Sende-Fehler. Reset.");
            },
            ontimeout: () => {
                isSending = false;
                lastSentHash = "";
                console.warn("[NEXUS Grok] Timeout. Reset.");
            }
        });
    };

    const cleanText = t => {
        let text = t.trim();
        if (text.includes("Grok Nexus Eye") || text.includes("nexus-input-target")) return "";
        text = text.replace(/_{3,}/g, '');
        text = text.replace(/\/{3,}/g, '');
        const lightTrash = [/This chat is private/i, /Generated by Grok/i, /xAI • /i];
        lightTrash.forEach(regex => { text = text.replace(regex, ''); });
        return text.replace(/\s+/g, ' ').trim();
    };

    const harvest = () => {
        if (isPaused) return;
        const containers = document.querySelectorAll('div.css-175oi2r[style*="display: block"]');
        if (containers.length === 0) return;
        const latest = containers[containers.length - 1];
        if (!latest || latest.querySelector('textarea') || latest.style.textAlign === 'right') return;

        const isFinished = document.querySelector('[aria-label*="Like"], [aria-label*="Share"], [aria-label*="Copy"], [aria-label*="Kopieren"], [aria-label*="Gefällt mir"]');

        if (!isFinished) {
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(harvest, 1000);
            return;
        }

        let clone = latest.cloneNode(true);
        const codeBlocks = clone.querySelectorAll('[data-testid="markdown-code-block"]');
        if (codeBlocks.length > 0) {
            codeBlocks.forEach(block => {
                block.outerHTML = " [System-Rauschen entfernt] ";
            });
        }

        let rawText = clone.innerText || clone.textContent || "";
        const cleaned = cleanText(rawText);
        if (cleaned.length >= 5) {
            send(cleaned);
            console.log("[NEXUS] Ernte abgeschlossen (Code isoliert)");
        }
    };


    const getThrottleMode = () => {
        const idleMs = Date.now() - lastActivity;
        if (idleMs >= IDLE_LIMIT) return 'deep_sleep';
        if (document.hidden || idleMs >= DROSS_LIMIT) return 'standby';
        return 'active';
    };

    const updateActivity = () => {
        lastActivity = Date.now();
        if (getThrottleMode() !== 'deep_sleep') startObserver();
    };

    const startObserver = () => {
        if (mutationObserver) return;
        mutationObserver = new MutationObserver(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') { stopObserver(); return; }
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(harvest, mode === 'standby' ? 10000 : PATIENCE_TIME);
        });
        mutationObserver.observe(document.body, { childList: true, subtree: true, characterData: true });
        isPaused = false;
        console.log("[Grok] Observer im Dienst.");
    };

    const stopObserver = () => {
        if (mutationObserver) {
            mutationObserver.disconnect();
            mutationObserver = null;
        }
        isPaused = true;
        console.log("[Grok] Tiefschlaf aktiviert.");
    };

    const startIdleChecker = () => {
        if (idleCheckerInterval) return;
        idleCheckerInterval = setInterval(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') stopObserver();
            else if (!mutationObserver) startObserver();
        }, 30000);
    };

    const wakeUpCall = () => {
        lastActivity = Date.now();
        if (mutationObserver) {
            mutationObserver.disconnect();
            mutationObserver = null;
        }
        startObserver();
        setTimeout(() => {
            console.log("[GROK] Manueller Synchronisations-Scan...");
            harvest();
        }, 500);
        console.log("[GROK] Observer reaktiviert!");
    };

    document.addEventListener('visibilitychange', () => {
        if (!document.hidden) wakeUpCall();
        else { isPaused = true; }
    });

    document.addEventListener('mousedown', () => {
        if (getThrottleMode() !== 'active') wakeUpCall();
        else updateActivity();
    });

    document.addEventListener('keydown', updateActivity);

    const markInput = () => {
        const input = document.querySelector('textarea[placeholder*="Ask anything"]');
        if (input && input.id !== "nexus-input-target") {
            input.id = "nexus-input-target";
        }
    };

    window.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            if (getThrottleMode() !== 'active') wakeUpCall();
            const input = document.getElementById('nexus-input-target');
            if (input) input.focus();
        }
    });

    setTimeout(() => {
        startObserver();
        startIdleChecker();
        setInterval(markInput, 2000);
        console.log("[Grok Nexus Eye v2.9.1] Titan-Wach-Modus bereit.");
    }, 3000);

})();
