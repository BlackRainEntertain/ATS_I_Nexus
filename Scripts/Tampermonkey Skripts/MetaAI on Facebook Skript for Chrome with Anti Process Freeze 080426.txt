// ==UserScript==
// @name         Meta Firefox Perfekt 070426 experimental anti process freeze
// @version      1.0
// @description  Extrahiert Meta AI Chat-Antworten für die barrierefreie Audio-Resonanz im Nexus-System.
// @author       Architekt & Gee
// @match        https://www.facebook.com/*
// @match        *://www.facebook.com*
// @match        *://www.meta.ai/*
// @match        *://www.messenger.com*
// @grant        GM_xmlhttpRequest
// @grant        unsafeWindow
// @run-at       document-end
// @connect      localhost
// @connect      *
// @connect      127.0.0.1
// @noframes     false
// ==/UserScript==


(function() {
    'use strict';
    const ENDPOINT = "http://127.0.0.1:8002/";
    let lastSent = "";
    let debounceTimer = null;

        // --- NEU: DER META-FOKUS-MAGNET (v2.0) ---
    window.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            // Wir suchen den Lexical-Editor-Container
            const metaInput = document.querySelector('[data-lexical-editor="true"]');
            if (metaInput) {
                metaInput.focus();
                console.log("[NEXUS] Meta-Fokus magnetisch auf Lexical-Editor gezogen!");
                // Visuelles Feedback
                metaInput.style.border = "2px solid #0084ff";
                setTimeout(() => { metaInput.style.border = "none"; }, 500);
            }
        }
    });

    // --- LARYNX-PARAMETER (Cyberpunk-Safe) ---
    const PATIENCE_TIME = 4000;
    const DROSS_LIMIT = 300000; // 5 Min Standby
    const IDLE_LIMIT = 900000; // 15 Min Deep Sleep
    let lastActivity = Date.now();
    let mutationObserver = null;
    let idleCheckerInterval = null;

    const getThrottleMode = () => {
        const idleMs = Date.now() - lastActivity;
        if (idleMs >= IDLE_LIMIT) return 'deep_sleep';
        if (document.hidden || idleMs >= DROSS_LIMIT) return 'standby';
        return 'active';
    };

    const updateActivity = () => {
        lastActivity = Date.now();
        if (getThrottleMode() !== 'deep_sleep') startObserver();
    };

    const harvestMeta = () => {
        const elements = document.querySelectorAll('p, div[dir="auto"], li, [role="article"]');
        let storyFragments = [];
        for (let i = elements.length - 1; i >= 0; i--) {
            const el = elements[i];
            const style = window.getComputedStyle(el);
            if (style.textAlign !== "right" && !el.isContentEditable && !el.closest('[role="textbox"]')) {
                const group = el.closest('[role="row"], [data-testid="message_container"], .x1n2onr6, [role="article"]');
                if (group) {
                    group.querySelectorAll('p, div[dir="auto"], li').forEach(p => {
                        const txt = p.innerText.trim();
                        if (txt && !storyFragments.includes(txt)) storyFragments.push(txt);
                    });
                } else { storyFragments.push(el.innerText.trim()); }
                break;
            }
        }
        const fullText = storyFragments.join("\n");
        if (fullText && fullText.trim() !== lastSent && !fullText.endsWith("...")) {
            lastSent = fullText.trim();
            send(fullText);
        }
    };

    const startObserver = () => {
        if (mutationObserver) return;
        mutationObserver = new MutationObserver(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') { stopObserver(); return; }
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(harvestMeta, (mode === 'standby' ? 10000 : PATIENCE_TIME));
        });
        mutationObserver.observe(document.body, { childList: true, subtree: true });
        console.log("[NEXUS Meta] Observer aktiv.");
    };

    const stopObserver = () => {
        if (mutationObserver) {
            mutationObserver.disconnect();
            mutationObserver = null;
        }
        console.log("[NEXUS Meta] Observer gestoppt (Deep-Sleep)");
    };

    const startIdleChecker = () => {
        if (idleCheckerInterval) return;
        idleCheckerInterval = setInterval(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') stopObserver();
            else startObserver();
        }, 30000);
    };

    function send(t) {
        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({ text: t, ts: Date.now() }),
            headers: { "Content-Type": "application/json" },
            timeout: 5000,
            onload: () => console.log("[NEXUS OK]"),
            onerror: (err) => console.error("[NEXUS ERR]", err)
        });
    }

    document.addEventListener('visibilitychange', updateActivity);
    document.addEventListener('keydown', updateActivity);
    document.addEventListener('mousedown', updateActivity);

    startObserver();
    startIdleChecker();
})(); // Ende der IIFE
