// ==UserScript==
// @name         GPT AI Nexus Eye (vivaldi) 22062026
// @version      1.0
// @description  Extrahiert ChatGPT Antworten und sendet sie an den lokalen GPT-Nexus (Port 8003)
// @author       Bre/Kairos & Gee
// @match        https://chatgpt.com*
// @grant        GM_xmlhttpRequest
// @connect      127.0.0.1
// @run-at       document-end
// ==/UserScript==

(function () {
    'use strict';

    // 1. HARDWARE- & NETZWERK-ANKER
    const TARGET_PORT = "8003";
    const ENDPOINT = `http://127.0.0.1:${TARGET_PORT}/webhook`;
    const MIN_LENGTH = 15;

    // 2. DROSSLUNGS-PARAMETER (IDENTISCH MIT GEE)
    const PATIENCE_TIME = 3500;
    const DROSS_LIMIT = 5 * 60 * 1000;
    const IDLE_LIMIT = 15 * 60 * 1000;

    // 3. SYSTEM-STATUS
    let lastActivity = Date.now();
    let lastSentHash = "";
    let debounceTimer = null;
    let mutationObserver = null;
    let idleCheckerInterval = null;
    let isPaused = false;
    let isSending = false;

    const simpleHash = str => {
        let h = 0;
        for (let i = 0; i < str.length; i++) h = ((h << 5) - h + str.charCodeAt(i)) | 0;
        return h.toString(36);
    };

    const send = (text) => {
        if (isSending || !text || text.length < MIN_LENGTH) return;

        // Titan-Hash: Ignoriert Interface-Änderungen nach 100 Zeichen
        const currentHash = simpleHash(text.slice(0, 100));
        if (currentHash === lastSentHash) return;

        isSending = true;
        lastSentHash = currentHash;

        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({
                text: text,
                ts: Date.now(),
                hash: currentHash,
                source: "gpt"
            }),
            headers: { "Content-Type": "application/json; charset=UTF-8" },
            timeout: 7000,
            onload: () => {
                setTimeout(() => { isSending = false; }, 1000);
                console.log("[GPT-Nexus] Audio-Resonanz stabil.");
            },
            onerror: () => { isSending = false; },
            ontimeout: () => { isSending = false; }
        });
    };

    const cleanText = t => {
        let text = t.trim();
        if (text.includes("ChatGPT") || text.includes("nexus-input-target")) return "";
        text = text.replace(/_{3,}/g, '');
        text = text.replace(/\/{3,}/g, '');

        // Bereinigung von verstecktem Interface-Müll aus dem DOM
        const lightTrash = [/Antwortaktionen/i, /Antwort kopieren/i, /Weitergeben/i, /^ChatGPT:/];
        lightTrash.forEach(regex => { text = text.replace(regex, ''); });
        return text.replace(/\s+/g, ' ').trim();
    };

    const harvest = () => {
        if (isPaused) return;

        // Nutzt den von dir gescouteten, tiefen Text-Container für das Fleisch der Nachricht
        const containers = document.querySelectorAll('div[data-message-author-role="assistant"]');
        if (containers.length === 0) return;
        const latest = containers[containers.length - 1];
        if (!latest || latest.querySelector('textarea')) return;

        // Der absolute Türsteher: Wartet auf das Erscheinen des Kopieren-Buttons unter der Antwort
        const isFinished = document.querySelector('[data-testid="copy-turn-action-button"], [aria-label="Antwort kopieren"]');

        if (!isFinished) {
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(harvest, 1000);
            return;
        }

        let clone = latest.cloneNode(true);

        // Schutz-Schelle: Ersetzt Programmiercode-Blöcke durch sauberes Systemrauschen
        const codeBlocks = clone.querySelectorAll('pre, code, .markdown prose');
        if (codeBlocks.length > 0) {
            codeBlocks.forEach(block => {
                block.outerHTML = " [System-Rauschen entfernt] ";
            });
        }

        let rawText = clone.innerText || clone.textContent || "";
        const cleaned = cleanText(rawText);
        if (cleaned.length >= 5) {
            send(cleaned);
            console.log("[NEXUS] GPT-5.5 Ernte abgeschlossen.");
        }
    };

    const getThrottleMode = () => {
        const idleMs = Date.now() - lastActivity;
        if (idleMs >= IDLE_LIMIT) return 'deep_sleep';
        if (document.hidden || idleMs >= DROSS_LIMIT) return 'standby';
        return 'active';
    };

    const updateActivity = () => {
        lastActivity = Date.now();
        if (getThrottleMode() !== 'deep_sleep') startObserver();
    };

    const startObserver = () => {
        if (window.nexusObserverGPT) {
            try { window.nexusObserverGPT.disconnect(); } catch(e) {}
            window.nexusObserverGPT = null;
        }

        if (mutationObserver) return;

        mutationObserver = new MutationObserver(() => {
            clearTimeout(debounceTimer);
            // 5 Sekunden Beruhigungszeit für flüssiges Rendern
            debounceTimer = setTimeout(() => {
                console.log("[GPT] Strom stabilisiert. Analysiere Fluss...");
                harvest();
            }, 5000);
        });

        mutationObserver.observe(document.body, { childList: true, subtree: true, characterData: true });
        window.nexusObserverGPT = mutationObserver;
        isPaused = false;
        console.log("[GPT-Nexus] Larynx-Observer im Dienst.");
    };

    const stopObserver = () => {
        if (mutationObserver) {
            mutationObserver.disconnect();
            mutationObserver = null;
            window.nexusObserverGPT = null;
        }
        isPaused = true;
        console.log("[GPT-Nexus] Drosselung: Tiefschlaf aktiv.");
    };

    const startIdleChecker = () => {
        if (idleCheckerInterval) return;
        idleCheckerInterval = setInterval(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') {
                stopObserver();
            } else if (document.hidden) {
                harvest();
                if (!mutationObserver) startObserver();
            }
        }, 10000);
    };

    const wakeUpCall = () => {
        lastActivity = Date.now();
        isPaused = false;
        if (window.nexusObserverGPT) {
            try { window.nexusObserverGPT.disconnect(); } catch(e) {}
            window.nexusObserverGPT = null;
        }
        mutationObserver = null;
        startObserver();
        console.log("[GPT] Observer reaktiviert.");
    };

    // TAB-WECHSEL & INTERAKTIONSMESSUNG
    document.addEventListener('visibilitychange', () => { if (!document.hidden) wakeUpCall(); });
    window.addEventListener('focus', wakeUpCall);

    document.addEventListener('mousedown', () => {
        if (getThrottleMode() !== 'active') wakeUpCall();
        else lastActivity = Date.now();
    });

    document.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            if (getThrottleMode() !== 'active') wakeUpCall();
            const input = document.getElementById('nexus-input-target');
            if (input) input.focus();
            return;
        }
        if (getThrottleMode() !== 'active') wakeUpCall();
        else lastActivity = Date.now();
    });

    const markInput = () => {
        // Findet das neue contenteditable-Eingabefeld über die verbliebene ID
        const input = document.querySelector('#prompt-textarea');
        if (input && input.id !== "nexus-input-target") {
            input.id = "nexus-input-target";
        }
    };

    setTimeout(() => {
        startObserver();
        startIdleChecker();
        setInterval(markInput, 2000);
        console.log("[GPT Nexus Eye v3.5] Titan-Wach-Modus auf Vivaldi bereit.");
    }, 3000);
})();
