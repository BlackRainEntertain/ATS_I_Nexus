// ==UserScript==
// @name         Ask_Studio_Nexus_Eye (v1.0) Chrome with Anti Process Freeze
// @version      1.0
// @description  Ask Studio monkey for Chrome
// @author       BlackRainEntertain
// @match        https://studio.youtube.com*
// @match        *://studio.youtube.com*
// @match        *://apis.google.com*
// @match        *://www.google.com*
// @include      *://studio.youtube.com*
// @grant        GM_xmlhttpRequest
// @connect      127.0.0.1
// @connect      127.0.0.1:8005
// @run-at       document-start
// @noframes     false
// ==/UserScript==

(function() {
    'use strict';

    const TARGET_PORT = "8005";
    const ENDPOINT = `http://127.0.0.1:${TARGET_PORT}/webhook`;
    const DROSS_LIMIT = 300000; // 5 Min Standby
    const IDLE_LIMIT = 900000; // 15 Min Deep Sleep

    let lastSentHash = "";
    let debounceTimer = null;
    let lastActivity = Date.now();
    let mutationObserver = null;
    let idleCheckerInterval = null;

    const simpleHash = str => {
        let h = 0;
        for (let i = 0; i < str.length; i++) h = ((h << 5) - h + str.charCodeAt(i)) | 0;
        return h.toString(36);
    };

    const getThrottleMode = () => {
        const idleMs = Date.now() - lastActivity;
        if (idleMs >= IDLE_LIMIT) return 'deep_sleep';
        if (document.hidden || idleMs >= DROSS_LIMIT) return 'standby';
        return 'active';
    };

    const updateActivity = () => {
        lastActivity = Date.now();
        if (getThrottleMode() !== 'deep_sleep') startObserver();
    };

    // Dein Spezial-Tool für YouTube Shadow-Roots
    const getDeepText = (node) => {
        let text = "";
        if (node.nodeType === Node.TEXT_NODE) return node.textContent;
        if (node.shadowRoot) text += getDeepText(node.shadowRoot);
        if (node.childNodes) {
            for (const child of node.childNodes) text += getDeepText(child);
        }
        return text;
    };

    const harvestStudio = () => {
        // Eingabefeld markieren
        const inputField = document.querySelector('.ytcpCreatorChatEntityAttachmentInlineFlowPromptBox[contenteditable="true"]');
        if (inputField && inputField.id !== "nexus-input-target") {
            inputField.id = "nexus-input-target";
        }

        // Antwort-Items scannen
        const items = document.querySelectorAll('ytcp-creator-chat-response-item, ytcp-ask-studio-item-view-model');
        if (items.length > 0) {
            const latest = items[items.length - 1];
            const rawText = getDeepText(latest);
            const cleaned = rawText.replace(/Kopieren|Feedback geben|Teilen/g, '').trim();

            if (cleaned.length > 10) {
                const hash = simpleHash(cleaned);
                if (hash !== lastSentHash) {
                    lastSentHash = hash;
                    send(cleaned);
                }
            }
        }
    };

    const startObserver = () => {
        if (mutationObserver) return;
        mutationObserver = new MutationObserver(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') { stopObserver(); return; }
            clearTimeout(debounceTimer);
            const delay = (mode === 'standby') ? 10000 : 2500;
            debounceTimer = setTimeout(harvestStudio, delay);
        });
        mutationObserver.observe(document.body, { childList: true, subtree: true });
        console.log("[Studio-Eye] Larynx aktiv.");
    };

    const stopObserver = () => {
        if (mutationObserver) { mutationObserver.disconnect(); mutationObserver = null; }
        console.log("[Studio-Eye] Deep-Sleep.");
    };

    const startIdleChecker = () => {
        if (idleCheckerInterval) return;
        idleCheckerInterval = setInterval(() => {
            getThrottleMode() === 'deep_sleep' ? stopObserver() : startObserver();
        }, 30000);
    };

    function send(t) {
        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({ text: t, owner: "Studio", ts: Date.now() }),
            headers: { "Content-Type": "application/json" }
        });
    }

    // Hotkey für Fokus
    window.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            const input = document.getElementById('nexus-input-target');
            if (input) input.focus();
        }
    });

    document.addEventListener('visibilitychange', updateActivity);
    document.addEventListener('keydown', updateActivity);
    document.addEventListener('mousedown', updateActivity);

    // Initialer Start nach 5 Sek (YT Studio braucht Ladezeit)
    setTimeout(() => {
        startObserver();
        startIdleChecker();
    }, 5000);
})();
