// ==UserScript==
// @name         GEE AI Nexus Eye from firefox for Chrome (STRUCTURAL v16) experimental
// @version      15.0
// @description  Extrahiert Google AI Overviews auch aus Shadow DOM und sendet reinen Text an localhost
// @match        *://www.google.de/*
// @match        *://www.google.com/*
// @match        *https://www.google.de/*
// @match        *https://www.google.com/*
// @grant        GM_xmlhttpRequest
// @connect      127.0.0.1
// @connect      127.0.0.1:8001
// @run-at       document-end
// ==/UserScript==

(function() {
    'use strict';

    const TARGET_PORT = "8001";
    const ENDPOINT = `http://127.0.0.1:${TARGET_PORT}/webhook`;
    const MIN_LENGTH = 65;
    const PATIENCE_TIME = 1500;

    let lastSentHash = "";
    let debounceTimer = null;
    let isInitialWait = true; // Blockiert nur die ersten Sekunden nach Laden

        const markInput = () => {
        const inputField = document.querySelector('textarea.ITIRGe');
        if (inputField && inputField.id !== "nexus-input-target") {
            inputField.id = "nexus-input-target";
            console.log("[NexusEye] Chat-Eingabe markiert (ID: nexus-input-target)");
        }
    };

    const simpleHash = str => {
        let h = 0;
        for (let i = 0; i < str.length; i++) h = ((h << 5) - h + str.charCodeAt(i)) | 0;
        return h.toString(36);
    };

const cleanText = t => {
    let text = t;
    // 1. Code-Blöcke weg
    text = text.replace(/`[\s\S]*?`/g, '');

    // Shadow-DOM Kleber vorab vernichten
    text = text.replace(/Vielen DankDein/g, '');
    text = text.replace(/Gute AntwortSchlechte Antwort/g, '');
    text = text.replace(/Feedback geben/g, '');
    text = text.replace(/KopierenTeilen/g, '');

    const trashWords = ["Kopieren", "Feedback", "Gute Antwort", "Schlechte Antwort", "Teilen", "Weitere Informationen"];


    // Wir suchen NUR im hinteren Bereich der Nachricht (die letzten 250 Zeichen)
    const searchAreaStart = Math.max(0, text.length - 250);
    const tail = text.substring(searchAreaStart);

    let cutPosInTail = -1;

    for (let i = 0; i < trashWords.length; i++) {
        let firstPos = tail.toLowerCase().indexOf(trashWords[i].toLowerCase());
        if (firstPos !== -1) {
            for (let j = 0; j < trashWords.length; j++) {
                if (i === j) continue;
                let secondPos = tail.toLowerCase().indexOf(trashWords[j].toLowerCase());
                // Wenn zwei Müllwörter im "Heck" nah beieinander liegen:
                if (secondPos !== -1 && Math.abs(firstPos - secondPos) < 100) {
                    cutPosInTail = Math.min(firstPos, secondPos);
                    break;
                }
            }
        }
        if (cutPosInTail !== -1) break;
    }

    // Wenn wir im Heck fündig wurden, schneiden wir das Original an dieser Stelle ab
    if (cutPosInTail !== -1) {
        text = text.substring(0, searchAreaStart + cutPosInTail);
    }

    // 2. Sonderzeichen-Schutz (*, (, ), [, ])
    text = text.replace(/[#\\{}]/g, '');

    return text.replace(/\s+/g, ' ').trim();
};




    const deepTextExtract = (node, depth = 0) => {
        if (depth > 12) return "";
        let text = "";
        if (node.nodeType === Node.TEXT_NODE) text += node.textContent;
        else if (node.nodeType === Node.ELEMENT_NODE) {
            if (["script", "style", "noscript"].includes(node.tagName.toLowerCase())) return "";
            if (node.shadowRoot) text += deepTextExtract(node.shadowRoot, depth + 1);
            for (const child of node.childNodes) text += deepTextExtract(child, depth + 1);
        }
        return text;
    };

    const send = (text) => {
        if (!text || text.length < MIN_LENGTH || isInitialWait) return;
        const hash = simpleHash(text);
        if (hash === lastSentHash) return;

        lastSentHash = hash;
        console.log("[NexusEye] Sende Resonanz → " + TARGET_PORT);

        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({ text: text, ts: Date.now(), hash: hash }),
            headers: { "Content-Type": "application/json; charset=UTF-8" }
        });
    };

    const harvest = () => {
        // Wir suchen jetzt gezielt nach dem neuen Container-Anker KFl8ub
        const elements = [...document.querySelectorAll('div[jsname="KFl8ub"]')];

        const aiMessages = elements.filter(el => {
            const style = window.getComputedStyle(el);
            // Wir filtern ALLES aus, was rechts steht oder editierbar ist
            const isUser = el.closest('[contenteditable="true"]') ||
                           style.textAlign === 'right' ||
                           style.textAlign === 'end';

            return el.innerText && el.innerText.length > MIN_LENGTH && !isUser;
        });

        const latest = aiMessages.slice(-1)[0]; // Das wirklich letzte Element
        if (latest) {
            const rawText = deepTextExtract(latest);
            const cleaned = cleanText(rawText);
            if (cleaned.length >= MIN_LENGTH) send(cleaned);
        }
    };


    // Initialer Timer: Nach 5 Sekunden ist der Affe erst "scharf"
    setTimeout(() => {
        isInitialWait = false;
        console.log("[NexusEye] Initial-Sperre aufgehoben. Gee lauscht.");
    }, 5000);

    // --- NEU: Alle 2 Sekunden prüfen, ob das Eingabefeld da ist ---
    setInterval(markInput, 2000);

        // --- DER FOKUS-FÄNGER (Reagiert auf Python) ---
    window.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            const input = document.getElementById('nexus-input-target');
            if (input) {
                input.focus();
                console.log("[NexusEye] Fokus magnetisch angezogen!");
            }
        }
    });


    // Der Observer (Nur EINMAL deklarieren!)
    const observer = new MutationObserver(() => {
        clearTimeout(debounceTimer);
        debounceTimer = setTimeout(harvest, PATIENCE_TIME);
    });

    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
    console.log("[NexusEye v16.3] LARYNX-Anker aktiv.");
})();
