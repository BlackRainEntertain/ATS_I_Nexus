// ==UserScript==
// @name         Ai on Google Search - Nexus Eye v17 060426 Chrome von Grok überarbeitet
// @version      17.0
// @description  Extrahiert Google AI Overviews auch aus Shadow DOM und sendet reinen Text an localhost + ADAPTIVES LARYNX-UPGRADE (Deep-Sleep / CPU-Schutz)
// @match        *://www.google.de/*
// @match        *://www.google.com/*
// @match        *https://www.google.de/*
// @match        *https://www.google.com/*
// @grant        GM_xmlhttpRequest
// @connect      127.0.0.1
// @connect      127.0.0.1:8001
// @run-at       document-end
// ==/UserScript==

(function() {
    'use strict';

    const TARGET_PORT = "8001";
    const ENDPOINT = `http://127.0.0.1:${TARGET_PORT}/webhook`;
    const MIN_LENGTH = 65;
    const PATIENCE_TIME = 3000;

    // === NEU: ADAPTIVES LARYNX-UPGRADE (v17) ===
    let lastActivity = Date.now();
    let isPaused = false; // Deep-Sleep-Flag (CPU 0%)
    const DROSS_LIMIT = 5 * 60 * 1000; // 5 Minuten → Standby-Modus
    const IDLE_LIMIT = 15 * 60 * 1000; // 15 Minuten → Deep-Sleep

    let lastSentHash = "";
    let debounceTimer = null;
    let isInitialWait = true;

    let mutationObserver = null;
    let idleCheckerInterval = null;
    // ==========================================

    const markInput = () => {
        const inputField = document.querySelector('textarea.ITIRGe');
        if (inputField && inputField.id !== "nexus-input-target") {
            inputField.id = "nexus-input-target";
            console.log("[NexusEye] Chat-Eingabe markiert (ID: nexus-input-target)");
        }
    };

    const simpleHash = str => {
        let h = 0;
        for (let i = 0; i < str.length; i++) h = ((h << 5) - h + str.charCodeAt(i)) | 0;
        return h.toString(36);
    };

    const cleanText = t => {
        let text = t;

        // 0. VORREINIGUNG
        text = text.replace(/```(markdown|javascript|python|html|css|json)?/gi, '$1');

        // 1. DIE ROHRSCHELLE (Vakuum-Schnitt)
        const blockRegex = /(markdown|javascript|python|html|css|json)[\s\S]*?(Verwende Code mit Vorsicht|Use code with caution|Rechtsgründen|Nutzungsbedingungen|rechtlichen Problemen|Antrag auf Entfernung|Weitere Informationen)\.?/gi;
        text = text.replace(blockRegex, ' [System-Rauschen entfernt] ');

        // 2. DIE INNERE DICHTUNG (Vollständige Panzerung)
        const labelTrash = [
            /Vielen\s*Dank\s*Dein/gi,
            /Gute\s*Antwort\s*Schlechte\s*Antwort/gi,
            /Feedback\s*geben/gi,
            /Kopieren\s*Teilen/gi,
            /In\s*Zwischenablage\s*kopiert/gi,
            /Verwende\s*Code\s*mit\s*Vorsicht\.?/gi,
            /Kopieren\s*Öffentlicher\s*Link\s*wird\s*erstellt…/gi,
            /Eine\s*Kopie\s*dieses\s*Chats\s*wird\s*deinem\s*Feedback\s*beigefügt/gi,
            /Senden\s*Vielen\s*Dank\s*für\s*dein\s*Feedback/gi,
            /Google\s*kann\s*Konto-\s*und\s*Systemdaten\s*verwenden/gi,
            /Datenschutzerklärung\s*von\s*Google/gi,
            /Google-Nutzungsbedingungen/gi,
            /Antrag\s*auf\s*Entfernung\s*aus\s*Rechtsgründen\s*stellen/gi,
            /Bei\s*rechtlichen\s*Problemen\s*kannst\s*du\s*einen\s*Antrag/gi,
            /Rechtsgründen\s*stellen/gi,
            /Diese\s*Aussagen\s*dienen\s*lediglich\s*Informationszwecken/gi,
            /Für\s*medizinische\s*Beratung\s*oder\s*eine\s*Diagnose/gi,
            /solltest\s*du\s*dich\s*an\s*einen\s*Experten\s*wenden/gi,
            /KI-Antworten\s*können\s*Fehler\s*enthalten/gi,
            /Weitere\s*Informationen/gi,
            /um\s*dein\s*Feedback\s*nachzuvollziehen\s*und\s*seine\s*Dienste\s*zu\s*verbessern/gi,
            /Dabei\s*gelten\s*die\s*und\s*die/gi,
            /Google\s*kann\s*Konto-\s*und\s*Systemdaten\s*verwenden/gi,
            /,\s*\.\s*\.\s*Bei/gi,
            /,\s*\.\s*Dabei\s*gelten/gi,
            /\.\s*Bei\s*rechtlichen\s*Problemen/gi,
            /^\s*Bei\s*$/gm
        ];

        labelTrash.forEach(regex => { text = text.replace(regex, ''); });

        text = text.replace(/```markdown/gi, '');
        text = text.replace(/```/g, '');
        text = text.replace(/`/g, '');

        // --- 3. DIE HECK-REINIGUNG (Präzisions-Schere v4.0) ---
        const instantKill = ["rechtsgründen stellen", "rechtlichen problemen", "nutzungsbedingungen", "antrag auf entfernung", "medizinische beratung"];
        const comboKill = ["kopieren", "feedback", "teilen", "informationen", "experten", "diagnose"];

        const searchAreaStart = Math.max(0, text.length - 350);
        const tailLow = text.substring(searchAreaStart).toLowerCase();
        let cutPosInTail = -1;

        for (let word of instantKill) {
            let pos = tailLow.indexOf(word);
            if (pos !== -1) {
                cutPosInTail = pos;
                break;
            }
        }

        if (cutPosInTail === -1) {
            let hits = 0;
            let firstPos = -1;
            comboKill.forEach(word => {
                let pos = tailLow.indexOf(word);
                if (pos !== -1) {
                    hits++;
                    if (firstPos === -1 || pos < firstPos) firstPos = pos;
                }
            });
            if (hits >= 2) cutPosInTail = firstPos;
        }

        if (cutPosInTail !== -1) {
            text = text.substring(0, searchAreaStart + cutPosInTail);
        }

        // 4. FINALE GLYPHEN-POLITUR
        text = text.replace(/[#\\{}]/g, '');

        return text.replace(/\s+/g, ' ').trim();
    };

    const deepTextExtract = (node, depth = 0) => {
        if (depth > 12) return "";
        let text = "";
        if (node.nodeType === Node.TEXT_NODE) text += node.textContent;
        else if (node.nodeType === Node.ELEMENT_NODE) {
            if (["script", "style", "noscript"].includes(node.tagName.toLowerCase())) return "";
            if (node.shadowRoot) text += deepTextExtract(node.shadowRoot, depth + 1);
            for (const child of node.childNodes) text += deepTextExtract(child, depth + 1);
        }
        return text;
    };

    const send = (text) => {
        if (!text || text.length < MIN_LENGTH || isInitialWait) return;
        const hash = simpleHash(text);
        if (hash === lastSentHash) return;

        lastSentHash = hash;
        console.log("[NexusEye] Sende Resonanz → " + TARGET_PORT);

        GM_xmlhttpRequest({
            method: "POST",
            url: ENDPOINT,
            data: JSON.stringify({ text: text, ts: Date.now(), hash: hash }),
            headers: { "Content-Type": "application/json; charset=UTF-8" }
        });
    };

    const harvest = () => {
        const elements = [...document.querySelectorAll('div[jsname="KFl8ub"]')];

        const aiMessages = elements.filter(el => {
            const style = window.getComputedStyle(el);
            const isUser = el.closest('[contenteditable="true"]') ||
                           style.textAlign === 'right' ||
                           style.textAlign === 'end';

            return el.innerText && el.innerText.length > MIN_LENGTH && !isUser;
        });

        const latest = aiMessages.slice(-1)[0];
        if (latest) {
            const rawText = deepTextExtract(latest);
            const cleaned = cleanText(rawText);
            if (cleaned.length >= MIN_LENGTH) send(cleaned);
        }
    };

    // ==========================================
    // LARYNX-UPGRADE: Throttle-Logik (NICHTS an cleanText / deepTextExtract ändern!)
    // ==========================================
    const getThrottleMode = () => {
        const idleMs = Date.now() - lastActivity;
        const isHidden = document.hidden;

        if (idleMs >= IDLE_LIMIT) return 'deep_sleep';
        if (isHidden || idleMs >= DROSS_LIMIT) return 'standby';
        return 'active';
    };

    const updateActivity = () => {
        lastActivity = Date.now();
        const mode = getThrottleMode();
        if (mode !== 'deep_sleep') {
            startObserver();
        }
    };

    const startObserver = () => {
        if (mutationObserver) return;

        mutationObserver = new MutationObserver(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') {
                stopObserver();
                return;
            }

            clearTimeout(debounceTimer);
            const delay = (mode === 'standby') ? 10000 : PATIENCE_TIME;
            debounceTimer = setTimeout(harvest, delay);
        });

        mutationObserver.observe(document.body, { childList: true, subtree: true, characterData: true });
        isPaused = false;
        console.log(`[NexusEye v17] Observer aktiv (${getThrottleMode()}-Modus)`);
    };

    const stopObserver = () => {
        if (mutationObserver) {
            mutationObserver.disconnect();
            mutationObserver = null;
        }
        isPaused = true;
        console.log("[NexusEye v17] Observer gestoppt → Deep-Sleep (CPU 0%)");
    };

    const startIdleChecker = () => {
        if (idleCheckerInterval) return;
        idleCheckerInterval = setInterval(() => {
            const mode = getThrottleMode();
            if (mode === 'deep_sleep') {
                stopObserver();
            } else {
                startObserver();
            }
        }, 30000); // extrem leichtgewichtiger Check
        console.log("[NexusEye v17] Idle-Checker gestartet (30s)");
    };
    // ==========================================

    // Initialer Timer: Nach 5 Sekunden ist der Affe erst "scharf"
    setTimeout(() => {
        isInitialWait = false;
        console.log("[NexusEye v17] Initial-Sperre aufgehoben. Gee lauscht.");
    }, 5000);

    // --- NEU: Alle 2 Sekunden prüfen, ob das Eingabefeld da ist ---
    setInterval(() => {
        const inputField = document.querySelector('textarea.ITIRGe');
        if (inputField && inputField.id !== "nexus-input-target") {
            inputField.id = "nexus-input-target";
            console.log("[NexusEye] Chat-Eingabe markiert (ID: nexus-input-target)");
        }
    }, 2000);

    // --- DER FOKUS-FÄNGER (Reagiert auf Python) ---
    window.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyY') {
            const input = document.getElementById('nexus-input-target');
            if (input) {
                input.focus();
                console.log("[NexusEye] Fokus magnetisch angezogen!");
            }
        }
    });

    // === LARYNX-UPGRADE: Event-Listener für Activity + Wake-Up ===
    document.addEventListener('visibilitychange', () => {
        if (!document.hidden) {
            updateActivity();
            console.log("[NexusEye v17] Tab im Fokus → Wake-Up!");
        }
    });

    // Aktivität nur über keydown + mousedown im Tab (wie im Briefing gefordert)
    document.addEventListener('keydown', updateActivity);
    document.addEventListener('mousedown', updateActivity);

    // Starte alles
    startObserver();
    startIdleChecker();

    console.log("[NexusEye v17 LARYNX-UPGRADE] Adaptive Drosselung aktiv. Zombie-Prozesse während Gaming-Sessions eliminiert.");
})();
